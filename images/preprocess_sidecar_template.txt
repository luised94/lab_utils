# Preprocessing sidecar template - gel densitometry pipeline
#
# Copy this file once per scan, next to the input .tif, named
#   <input stem>_preprocess.txt
#
# THIS FILE IS THE AUTHORITY FOR THE KEY LIST. validate_gel_image.py duplicates
# the list rather than importing it (CONVENTIONS.md section 1 forbids sharing
# between scripts). A missing key and an unknown key are both hard stops, so
# drift between this file and the script fails on the next run instead of being
# silently ignored.
#
# Format:
#   - one key=value per line, no spaces around the =
#   - lines starting with # are comments, blank lines are ignored
#   - every key below is REQUIRED; the two free-text ones may be left empty
#
# WHAT FRAME YOU MEASURE IN. Open the raw .tif in Fiji and measure it exactly as
# it opens: no rotation, no flip, no save. Fiji is a ruler here, nothing else.
# The scan origin is bottom-left, so file row 0 is the physical BOTTOM of the gel
# and Fiji draws row 0 at the top of the window: the image is mirrored top-to-
# bottom on screen. You do not need to correct for this. Record what you see; the
# pipeline performs the flip later. The measured_in_frame key pins this contract.

schema_version=2

# Must be exactly raw_as_opened_unflipped. If the measuring convention ever
# changes, old sidecars fail loudly instead of being silently reinterpreted.
measured_in_frame=raw_as_opened_unflipped

# The filename these measurements were taken against, without any directory.
# Checked against the file actually being processed.
measured_against_input_filename=CHANGE_ME.tif

# Image dimensions, from Image > Properties or the window title bar. Checked
# against the TIFF tags. These keep the _pixels suffix because they are always
# pixels, read from the file, regardless of coordinate_unit below.
measured_against_image_width_pixels=0
measured_against_image_height_pixels=0

# The direction the samples migrate. REQUIRED, no default, so a gel imaged in
# either orientation cannot be processed under a silent assumption.
#   horizontal = samples run left-to-right; lanes are stacked top-to-bottom
#   vertical   = samples run top-to-bottom; lanes are stacked left-to-right
# Determine it by eye: find the wells (the loading points); migration leads away
# from them.
gel_migration_axis=CHANGE_ME

# The unit of every landmark_* and crop_* value below. REQUIRED.
#   pixels      = the raw pixel coordinates
#   centimetres = the calibrated values Fiji shows when the image carries a
#                 spatial scale; the pipeline converts them using the pixel size
#                 in the file's own resolution tag
# HOW TO TELL WHICH YOU HAVE: hover a pixel and read the status bar. If it shows
# something like  x=3.80 (476)  the image is calibrated: the leading number is
# centimetres and the number in parentheses is the pixel. If it shows a bare
# integer, you are already in pixels. Recommended: record pixels, using the
# getLine macro (see PROTOCOL.md) for the landmark, so nothing is rounded.
coordinate_unit=CHANGE_ME

# Two points on a line that is straight and LEVEL on the physical gel, used to
# derive the residual tilt. The tilt is derived, not typed, so the crop preview
# can draw the points back for you to check.
#   For gel_migration_axis=horizontal: use a line that runs top-to-bottom (the
#     well line is usually best). a = one end, b = the other end.
#   For gel_migration_axis=vertical:   use a line that runs left-to-right.
# Put a and b AS FAR APART AS YOU CAN along that line. One pixel of click error
# over a 950 px span is about 0.06 degrees; over 400 px it is 0.14. The tilt is
# the deviation of the a-b line from the axis perpendicular to migration.
landmark_a_x=0
landmark_a_y=0
landmark_b_x=0
landmark_b_y=0

# What you actually clicked, in words, and which end is a and which is b. This is
# the only record of your choice. May be empty, but an empty one costs you the
# ability to reproduce it.
rotation_landmark_description=

# Crop box as Fiji reports a rectangle selection: x, y, width, height of its
# top-left corner and size.
#
# THE CROP IS NOT A MEASUREMENT BOUNDARY; the per-band boxes set later are. Its
# only job is to enclose every band generously. The rule:
#   DO   contain every band you intend to measure, plus 20 to 50 pixels of clean
#        gel margin on all four sides: margin along the migration axis so each
#        band has background on both sides for its local baseline, and margin
#        along the lane-stacking axis so the outermost lanes are fully inside.
#   DO   err generous. Too large costs a slightly slower run. Too tight, clipping
#        a band or the clean gel flanking it, biases that lane consistently and
#        believably and wrongly.
#   DO NOT clip inward through any band or its flanking gel to tidy the edges or
#        to exclude bare plate.
# The wells (the loading origin) are a separate choice. If no measured band sits
# near them you may leave them just outside, which avoids pulling in a strip of
# bare plate and the near-saturated well artifact. If a band sits near them, or
# you are unsure, include them: a superset is always safe.
crop_x=0
crop_y=0
crop_width=0
crop_height=0

# Number of lanes you expect inside the crop, empty lanes included. Stage 1
# range-checks it and cross-checks the implied lane pitch (the center-to-center
# spacing between adjacent lanes) against the pixel size from the file.
expected_lane_count=0

# Anything worth remembering about this scan. May be empty.
notes=
